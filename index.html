fetchPreview = async (artist, song, existingUrl) => {
  if (existingUrl && existingUrl !== "NA" && existingUrl !== "") return existingUrl;
  if (!song || !artist) return null;
  try {
    const clean = song.replace(/\s*\(.*?\)/g, "").trim();
    const query = encodeURIComponent(`${artist} ${clean}`);
    const res = await fetch(`https://itunes.apple.com/search?term=${query}&entity=song&limit=1`);
    if (!res.ok) return null;
    const data = await res.json();
    return data.results?.[0]?.previewUrl || null;
  } catch (e) {
    return null;
  }
}
